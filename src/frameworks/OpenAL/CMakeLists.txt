project(OpenAL)

remove_sdk_framework(OpenAL)

set(DYLIB_COMPAT_VERSION "1.0.0")
set(DYLIB_CURRENT_VERSION "1.0.0")

set(FRAMEWORK_VERSION "A")

generate_sdk_framework(OpenAL
    VERSION ${FRAMEWORK_VERSION}
    HEADER "include/OpenAL"
)

# The framework is built from openal-soft (LGPL-2.1+; the framework is a
# dynamic library and its source is the openal-soft submodule) with its
# CoreAudio backend, which renders through AudioUnit. OpenAL sits on

set(OPENAL_SOFT_DIR "${CMAKE_SOURCE_DIR}/src/external/openal-soft")

set(openal_sources "")
foreach (src IN LISTS openal_soft_sources)
    list(APPEND openal_sources "${OPENAL_SOFT_DIR}/${src}")
endforeach()

# The resampler tables are generated by a tool that runs on the build machine,
# so it is compiled here with the unflagged compiler (like HOST_CC in
# src/CMakeLists.txt) instead of as a target for the Darwin build.
set(openal_gen_dir "${CMAKE_CURRENT_BINARY_DIR}/gen")
add_custom_command(
    OUTPUT "${openal_gen_dir}/bsinc_inc.h"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${openal_gen_dir}"
    COMMAND ${CMAKE_C_COMPILER} -O1 -o "${openal_gen_dir}/bsincgen"
        "${OPENAL_SOFT_DIR}/native-tools/bsincgen.c" -lm
    COMMAND "${openal_gen_dir}/bsincgen" "${openal_gen_dir}/bsinc_inc.h"
    DEPENDS "${OPENAL_SOFT_DIR}/native-tools/bsincgen.c"
    VERBATIM
)

add_framework(OpenAL
    FAT
    CURRENT_VERSION
    VERSION ${FRAMEWORK_VERSION}

    SOURCES
        ${openal_sources}
        "${CMAKE_CURRENT_BINARY_DIR}/gen/bsinc_inc.h"

    LINK_FLAGS
        "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/OpenAL.exp"

    DEPENDENCIES
        system
        cxx
        CoreAudio
        AudioToolbox
        CoreFoundation
)

target_include_directories(OpenAL BEFORE PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/config"
    "${CMAKE_CURRENT_BINARY_DIR}/gen"
    "${OPENAL_SOFT_DIR}/include"
    "${OPENAL_SOFT_DIR}/common"
    "${OPENAL_SOFT_DIR}/alc"
    "${OPENAL_SOFT_DIR}"
)
target_compile_definitions(OpenAL PRIVATE AL_BUILD_LIBRARY AL_ALEXT_PROTOTYPES)
set_target_properties(OpenAL PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON)
